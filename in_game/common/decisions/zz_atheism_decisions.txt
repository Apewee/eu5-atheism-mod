# Available to every Lutheran country: any of them can produce a thinker radical
# enough to do away with religion itself.
sponsor_a_radical_thinker = {
	decision_category = religious_decisions
	image = "gfx/interface/illustrations/decisions/atheism_radical_thinker.dds"

	potential = {
		religion = religion:lutheran
	}

	allow = {
		has_ruler = yes
		gold >= {
			value = monthly_income_trade_and_tax
			multiply = 6
		}
	}

	ai_will_do = {
		value = -100
	}

	option = {
		name = sponsor_a_radical_thinker.a
		ai_chance = {
			value = 0
		}

		effect = {
			add_gold = {
				value = monthly_income_trade_and_tax
				multiply = -6
			}

			# Atheism must exist as a religion before anyone can be converted to it.
			religion:atheism = {
				enable_religion = yes
			}

			create_character = {
				estate = estate_type:clergy_estate
				culture = root.culture
				religion = religion:atheism
				adm = 90
				dip = 70
				mil = 30
				save_scope_as = radical_thinker
			}

			change_religion = religion:atheism
			change_religion_for_ruler_and_family = {
				country = root
				religion = religion:atheism
			}

			# The thinker's circle becomes a movement: it spreads atheism through the
			# population on its own, the way the Lutheran and Hellenist movements do.
			capital = {
				spawn_movement = {
					movement_definition = movement_definition:atheism_movement
					supporters = {
						value = population
						multiply = 0.2
					}
				}
			}
		}
	}
}
