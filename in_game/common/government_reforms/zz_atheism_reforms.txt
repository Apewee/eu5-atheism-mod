# The Atheist state's defining reform: the state itself takes the place of the church.
# It grants the War of Reason casus belli against every revealed faith and a permanent
# trickle of research progress for the state's academies.
#
# Deliberately no 'age =' line: a country only ever becomes Atheist through the radical
# thinker decision, which can happen long before the Reformation age, so the reform has
# to be available as soon as the state has adopted Atheism. The religion check below is
# the only requirement.
atheism_supremacy_of_reason = {
	potential = {
		religion = religion:atheism
	}

	allow = {
	}

	country_modifier = {
		# +0.10 base research progress every month
		research_speed = 0.1
		stability_cost_efficiency = 0.05
	}

	on_activate = {
		unlock_casus_belli_effect = { type = cb_atheism_war_of_reason }
	}

	on_deactivate = {
		lock_casus_belli_effect = { type = cb_atheism_war_of_reason }
	}

	years = 2
}
